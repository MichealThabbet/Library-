الشهيدان قزمان ودميان-السلام للمختارين

وإخوتهم القدَّيسين

السلام للمُختارين

السلام للشهداء

وثيؤدوتى أمهم أجمعين

معناها عطية الله

ثيؤدوتى جبيبة الله

السلام للشهداء

علمتهُم خوف الله

وأخوهم إبرابيوس

أنتيموس ولاوندوس

السلام للشهداء

عاشوا فى حُب إيسوس

فى الطب بارعان

قزمان ودميان

السلام للشهداء

تَمجد الله الحنَّان

والمرضى مُتساويين

للمال كانا مُحتقرين

السلام للشهداء

وفى الإيمان ثابتين

خلَّصا من الشيطان

فتحا أعيُن العُميان

السلام للشهداء

بقوة الله الديان

وبالمسيح فرحوا

الوثنيون آمنوا

السلام للشهداء

وللإيمان نشروا

عذَّب عبيد الفادى

سمع بهم الوالى

السلام للشهداء

سفك دمهُم الغالى

صَلَّوا أصبح سالم

مرض الوالى الظالم

السلام للشهداء

لكن حقده دائم

ومار بقطر بحث عنهم

قطع رأس أمهم

السلام للشهداء

ودفنها ونال بركتهم

وهم يصرخون مُسبَّحين

سفك دم الباقين

السلام للشهداء

يا إلهنا ثبَّت المسيحيين

أصبح عيدهم مشهور

فى شهر هاتور

السلام للشهداء

نُعَّيد لهُم بسرور

أصبح لهم كنيسة أعجوبة

فى شهر بؤونة

السلام للشهداء

بالمعجزات مشهورة

السلام لكم يا أبرار

طُوباكم يا أطهار

السلام للشهداء

أكاليلكم ذهب مُختار

للمسيح نصير بنون

أطلبوا عنَّا يا قديسون

السلام للشهداء

وفى الملكوت نكون

كل المؤمنين

تسبيح اسمك فى أفواه

قزمان ودميان اعنَّا أجمعين

الكل يقولون يا إله
