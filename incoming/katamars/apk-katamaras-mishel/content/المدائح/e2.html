القديسة حنة والدة العذراء مريم

قديستنا الغالية

من عيلة نقية

تي أجيا حنة

حياتها نقية

و أمها مريم

إبنة لمثان

تي أجيا حنة

من نسل مُكرم

قوية و روحانية

أخوة حقيقية

تي أجيا حنة

ماريا وصوفيا

و نقاوة السريرة

في هدوء السيرة

تي أجيا حنة

كالشمعة المُنيرة

مع يوقيم زوجها

سارت بخوف الله

تي أجيا حنة

و كانت من الأمراء

في جمالها بارعة

عاشت بودعة

تي أجيا حنة

و روحها مُتضعة

و اعانت البؤساء

أحبت الفقراء

تي أجيا حنة

فرحت السماء

بخشوع أمام الله

حباتها صلاة

تي أجيا حنة

سالكة في وصاياه

لعدم نسلها

دخل الحزن قلبها

تي أجيا حنة

فصلت لربها

كثيرًا ببكاء

سجدت امام الله

تي أجيا حنة

و طلبت برجاء

هي و زوجها يواقيم

و كانت بإيمان عظيم

تي أجيا حنة

في وعود الله واثقين

و قال لهم بثبات

ظهر لهم الملاك

تي أجيا حنة

الله سامع الصلوات

في السماء من جوهر كريم

رأوا تاجًا عظيم

تي أجيا حنة

على رأسهما يقيم

بالبشرى البهية

و بشرهم ذكريا

تي أجيا حنة

ميلاد فخر البتولية

يا حنة و يواقيم

نسلكم سيكون عظيم

تي أجيا حنة

و قيد الله

و قدموا القربان

قلبهم بقى فرحان

تي أجيا حنة

على مذبح الغفران

و رزقوا جوهرة نقية

و زالت العقوبة

تي أجيا حنة

زينة كل البشرية

هي ام الأرباب

إنا طاهر مُختار

تي أجيا حنة

بها خلاص الأدهار

مريم من كل قلب

نذرت حنة للرب

تي أجيا حنة

قدمها بكل الحب

على زينة الفردوس

إئتمنها القدوس

تي أجيا حنة

العذراء أم إيسوس

في غاية الإيمان

راعتها بكل حنان

تي أجيا حنة

في مخافة الديان

قدمت فخر البنات

بعد ثلاث سنوات

تي أجيا حنة

للهيكل مع هبات

يفرح و شكر دائم

و اقامة الولائم

تي أجيا حنة

الله بحب

أن ياتي من السماء

و لما أراد الله

تي أجيا حنة

أختار الحمامة الحسنة

و وبرح قدسه زارها

تجسد منها

تي أجيا حنة

لتفرح بمسرة

لأم ابن الله

قد صرت أما

تي أجيا حنة

طوباك في أعلى سماه

لما حملتي

يا فرحة قلبكِ

تي أجيا حنة

في العذراء سيدتي

كرسي ابن الله الغالية

حملتي السماء الثانية

تي أجيا حنة

كرامتك في السماء عالية

و لا صارت السلامة

حملتي الحمامة

تي أجيا حنة

و زالت الغمامة

كلية الطهارة

حملني المَنارة

تي أجيا حنة

أمنا العذراء

و الله فرح قلبك

عظيم هو قدرك

تي أجيا حنة

قد بارك

بصلاتك ننال سلام

معنى إسمك حنان

تي أجيا حنة

أذكرينا على الدوام

للسماء مرتفعة

صلاتك

تي أجيا حنة

و سريعة الندهه

و بساطة مع إيمان

قلبكِ كله حنان

تي أجيا حنة

و وداعة كالحمامة

انتقلتى بسرور

  11 من هاتور

تي أجيا حنة

يا أم أم النور

ليكون الله معنا

أذكرينا يا

تي أجيا حنة

في عرسه يجمعنا

في صلاتك يا امينة

أذكرينا بأسمينا

تي أجيا حنة

لنصل للميت

لم نسك في صفاتك

لم نحيا كحياتكِ

تي أجيا حنة

في صلاتك

كل المؤمنين

تفسير اسمكِ في افواه

حنة و يواقيم أعنا أجمعين

الكل يقولون يا إله
