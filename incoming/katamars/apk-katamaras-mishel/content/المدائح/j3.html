والـرعــــاة الــرؤســــــــاء

من أفـــاضــل الأبـــــــاء

الـبابـا شـــنـودة المـعطــاء

زادوا الكــرســــى إرتــــقــاء

اخــــتـاره ربّ الــعـــــــلاء

كبولس الرسول والإناء

ويـــزداد للملـكــوت أبــنـــاء

للمسـكونة كـوكـب وضـــيـاء

كـاسـقـف لـلـديــر ومـــنـار

أنـبـا ثـاؤفـيلـس باقـتدار

كــابـــــن لـــــه و بـــــَــــــار

رهــبــنـه بـكـــل وقـــــــــــار

مـن ديــرنــا الســـــــريــان

 من أفـاضـــل الرهـبـــان

فـــضــــــائـــل الـبـســــتـــان

عــلّــم الـرهـــــبان بـحـــنـان

و وجـهـه يُشـــيــع ســـرور

كـكـاهـن شـخصه وقــور

كـقــــدوة لـــكـــل الـديـــــور

بإفـــــــراز يـمـنع شـــــــرور

لـيــس عــنــده تــفــضــيــل

أب اعــــتـراف جـــلـيـل

ويشــعـرهـــم بـحـب جــزيـل

يـقـابـل الـكــل بـتــهـــــلــيــل

كــــان يُــصـلـى بـحـــرارة

 سـكـن فـــترة بمغــارة

ليـقــود شـــعــبـه كـــمـنــارة

لـكــن الـــربّ اخــــــــتـــــاره

عـمـل وصـايــاه بخــضــوع

أحـب الــــربّ يــســـوع

يــرويــنـا و مـعــنـا جــمـوع

فـصـــــــيـره لـنــا يــــنـبــوع

لـنـشــر الإيـمــان القــــويـم

رســم اســقـفـاً للتعـلـيم

الأنبا كـــيـرلــس الـحـكــــيـم

بـيـــد الــبابــا الـعـــظـــــــيــم

والـمعـــاهــــد الــــديـنــــيـة

كـمـديـر للإكــلـيركـيـة

بـالـكــــــرازة الـشــــنـوديــة

فـصـبـغـهـا بـدمــغـة قــــويـة

مـثـل حـبيب جـرجـس رائـد

لمدارس الأحــد قــائـد

أعــطــاهــا إهتــــمــام زائــد

و رغـــــم مـــــرور شـــدائـد

اخــتارتــه لـلبـطـــريـركــية

العـــنـايــة الإلـهـــيــة

والـــقرعـــة الهــــيكـــــلـــية

تــــزكــــيــة الــرعـــــــيـة

يُــصـلـى عـنـا كـشــــفـــيـع

أصبح راعـياً للقـطـيـع

بـمـواعــظ صـــوتـه الـبـديـع

و يُـغـــذيـه بـمـا يـسـتـطـــيـع

مُـخـــــتـار مـن الــســـــماء

كـراعـــى لـلـرُعـــــــاه

فـى نـــــــــور مـــا أبـــهـــاه

ظــهــرت والــــــدة الإلـــــــه

فى كـنائـس فى بحـر سـنين

تجـلـت أم السـمائـيـين

فى عـهـــد البـابـا الأمـــــيـن

و ظـهرت أعـضــاء لقديسين

كـمارمـرقـس عـبد إيـسوس

نفـذ مـشـيئة الـقـدوس

و أسـاقـفة لـرعـاية النفـوس

فــرســم مـــئـات قـــســــوس

فـى أمــاكــــن جــــــديـــــدة

  بنيت كنائــس عــدــيد

بـحـكــمــتــة الـــرشـــــــيــدة

وإيـــبـاروشـــيـات مـــزيـــدة

عـــــمّـــــر ديــــــــــــــارات

علـى مـدى ســـنوات

فــى جـــمـــيـــع الـــقـــارات

بـكـنـائــــس لـهـا مـــنـــارات

أرسـل أسـاقــفــة لــديـــوره

فـى أنحاء الـمعــمورة

فـتـمـجّــــد الــــربّ بــنـــوره

كـرائحـة المـسـيح و بـخـوره

فـــكـــر بــابــانــا الـمـبـرور

لأديرة إندثرت لعصور

والـجـديـدة أحـاطـها بـســور

فعـّـمـرهـا وخـرجـت لـلـنــور

فـكــتــب كـــتـب و أشـعـــار

لعقـله المسـيـح أنـــار

و تــعــالــيــم كــالأنـــــهـــار

وعـــظـــات كـــالأبـــــــــرار

لـيـفــتـقـد فــيـها العِـــــبــاد

زار الأقــطـار و الـبـلاد

فحُـسـب لــه إكــلـيل وجـهـاد

رغـــم أمـــراضــه كالأجــــداد

بـحـكــمـة وقـــلـب مــُـريــح

يرعـى قطـيع المسـيح

اســـتـحــق كــل مــــديـــــح

و بـلـســـان فـمـه الفــصــيـح

بـقـــلــــب لــــه صـــــبـــور

عـلى الإيـمـان غـيـور

بـــدون جــــرح  الـشــعـــور

يُـظـهـر الحـقــيــقــة كـالـنور

و كــداود بـمـــقــــلاعـــــــه

كأسـد فـى الشـجاعـة

و يحـطـم الشــوك بأنـواعـــه

يجـمـع الأســاقـفة بإجتماعـه

بـحـكــمـة و عــقـل نـفــيس

يـهـزم حــيـل إبـلـيـس

يعـطـى الإنجـــيـل الـتـقـديـس

و كـأب لـلكنـيـسـة و رئـيـس

و الـمـلائــكـة بــه تـحـــيــط

شـفيع لشعبه كوسـيط

كــراهـــب مـن الإســــقــــيـط

طــــعـامـه دائـــمـاً بـســـــيـط

فـى نــفـــوس عِـــــطــــاش

وعـظـه قـــد عـــــاش

يسـمعـه الــكُــل بانــدهــــاش

كــمُـعــلــــم غــيـر غـــــــاش

رأيــه دائــــمـاً مـطـــــلــوب

معروف لكل الشـعـوب

مـن الآخـــــريـــن مـحــبـوب

يـمــــلـــك عــلى الــقــلــــوب

لــلـكــهــنــة أب حــبــــيـــب

للأســاقـفة دائماً قريب

لـلشـــعـــب راعـــى رقــــيـب

للشمامسـة قـلـــب رحـــيـــب

إحــتـضـــن الــمـنـــبـوذيــن

فــــك الـمـأســـــوريـن

ردّ الـــــضـــــــــــالـــــــيــــن

خـــلــــــّص الــخــــاطــــئـيـن

لـكــل فـــقــيــر ويــتــــيــــم

عـطـف بقلـب رحـــيـم

لـلـمـريــض كـأخ كـــــريـــــم

كـصـــديــق لـهـــم قــــديـــــم

فـى زواج الـــــبـــــنـــــــات

أزال الـعـــقــــــــبـــات

فـإســتـحـق لــهـذا دعــــوات

بحـــــــل الـمــــشـــكـــــــلات

يـدبـر مـعـيـشـتها الحـقـيـرة

يعول الأســر الفـقـيرة

بـمـشــروعــاتـه الصـــغـيـرة

بـرحـــمـة قـلــبـه الـكـــبـيـرة

و النـعـمة على قـلبه تحـوز

الضيقة بصلاته تجـوز

تُـسـجّــل لـلـتـاريـخ كــرمـوز

و الـكنيســة بحكـمــته تـفـوز

وعِــلــو الــروحــــــــيــــات

جــمــع مـن الـقـامـات

و مــقـــاومـــة الهرطـــقــات

الــتــوحّــــــــد والــعــــظــات

و شـــــهـادات إمـــــتــيـازه

و البــركـة و الـكـرازة

و ســــــــمـــو طِــــــــــــرازه

فــضـــلاً عــن إفــــــــــــرازه

وسُــتـر مـن الإنــجـــــازات

ما خُفى من المعجزات

لــظـــــهــــوره أوقـــــــــــات

وســــمــو الــروحــــــــيـــات

نذكره بأجمل نشيد

لعـهــــده الـفـــــريــــد

كـبـابـا لـلـكـــرازة مـجــــــيـد

و نـمـدحــه دائـــماً بـتمجيد

بالحقيقة قــولـــوا آمـــــيـن

نياحاً للبابا الأمين

بصـلواتـه عـــنـا أجـمـعـــيـن

و يــنــفـعـنـا كــل حـــــــيـــن

 الكل يقولون يا اله البابا شتودة اعنا اجمعين

خين إفران إم إفيوت نيم إبشيرى نيم بى ابنفما إثؤاواب تى إترياس إثؤواب إن أومو أو سيوس
