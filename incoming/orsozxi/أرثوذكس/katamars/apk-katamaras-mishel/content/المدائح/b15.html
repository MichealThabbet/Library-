الشهيد البابا بطرس خاتم الشهداء-أفتح فاي بالتسبيح

لربي يسوع المسيح

افتح فاي بالتسبيح

القديس أنبا بطرس

وبكل قواي أصيح

الي حبيب مسيحي

أوجه مديحي

القديس أنبا بطرس

من نال التجريح

الابرار الانقياء

هو خاتم الشهداء

القديس أنبا بطرس

مكانه في السماء

من بين الباباوات

وهو السابع عشر

القديس أنبا بطرس

بايمانه الواضح ظهر

كاهن بار وقديس

والد هذا الطاهر

القديس أنبا بطرس

قي الاسكندرية ظاهر

ولد يقر عينه

ولم يكن لوالده

القديس أنبا بطرس

فصلي الي ربه

ذهبت أم القديس

في الخامس من أبيب

القديس أنبا بطرس

الي كنيسة المهيب

حاملات أولادهن ببهاء

ورأت فيها النساء

القديس أنبا بطرس

فبكيت الي السماء

رسولا ربها

في الليل ظهرا لها

القديس أنبا بطرس

بطرس وبولس محبيها

أن الرب الأمين

وأعلماها بيقين

القديس أنبا بطرس

سيرزقها ولد ثمين

بطرس ومضت كقوله

وسوف يكون اسمه

القديس أنبا بطرس

الي رئيس شعبه

بالرؤية ووضحت

زوجها عرفت

القديس أنبا بطرس

ففرح بما رأي

طلب لها معونة

والبابا ثاؤنا

القديس أنبا بطرس

من رب المسكونة

أن رزقت بها الجليل

وحدث بعد قليل

القديس أنبا بطرس

فأسمته بطرس بتهليل

سلموه بيقين

في كمال سبع سنين

القديس أنبا بطرس

الي البابا الثمين

اللاهوتية المقدسة

فألحقه بالمدرسة

القديس أنبا بطرس

فصار حجة خالصة

والوعظ والارشاد

وبرع في الاجتهاد

القديس أنبا بطرس

وتكرس لرب العباد

ثم قسا للقداس

أغنسطس فشماس

القديس أنبا بطرس

فحمل باحتراس

كثيرا من شئون

عن البابا المصون

القديس أنبا بطرس

البيعة بفنون

البابا ثاؤنا المهاب

ثم تنيح يا أحباب

القديس أنبا بطرس

فجلس ها الأب

بعلومه النفيسة

فاستضاءت الكنيسة

القديس أنبا بطرس

وصلاته الوديعة

من بعد ما نصحه

آريوس المخالف قطعه

القديس أنبا بطرس

للوالي رفع أمره

وقتلوا من شعبه

فأمر بقطع رأسه

القديس أنبا بطرس

كثيرين له شهدوا

أن يسلم نفسه باشهاد

فرأي فخر العباد

القديس أنبا بطرس

عن شعبه باجتهاد

وطلب منه بنقاء

وصلي لرب السماء

القديس أنبا بطرس

أن يكون آخر الشهداء

يقولون له آمين

فسمع صوت حنين

القديس أنبا بطرس

ففرح هذا الثمين

تقدم لسيافه

لما أتم صلاته

القديس أنبا بطرس

فقطع بعنف رأسه

كل المؤمنين

تفسير اسمك في أفواه

البابا بطرس أعنا أجمعين

الكل يقولون يا اله
