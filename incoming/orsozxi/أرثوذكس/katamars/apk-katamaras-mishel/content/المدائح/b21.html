القديس الأنبا بولا أول السواح-يا أنبا بولا

يا اول السواح

يا انبا بولا

بتنور كالمصباح

يا قديس وبار

جيت هنا بسماح

سيرتك نقية

وقلبك بيه ارتاح

من رب القوات

ولا بصيت وراك

سيبت العالم كله

بالرب اللى فداك

وقلبك كان مشغول

دا يسوع هو اللى هداك

مين زيك يا حبيب

وحدك ولا حد معاك

تسكن ده المكان

مع كل غروب غراب

رب القدرة يامر

لا اتاخر يوم ولا غاب

معاه نصف خبزة

تحتار فيه الالباب

ده امر غريب وعجيب

لا ده رب الارباب

طير بيعول انسان

قرار

يا حبيب المسيح

طوباك يا انبا بولا

وجهك حلو ومريح

يا ضياء منير

تشفى القلب الجريح

صلواتك بلسم

من كل كلام فصيح

وصمتك اقوى

والمال كان عنده كثير

لما مات ابوه

وقال له انت صغير

طمع فيه اخوه

ونصيبى انا الكبير

نصيبك يبقى قليل

اللى لازم تصير

هى دى القسمة

من اللى ظلموه

حب ينصفوه

يحكم ما بينه و بينه

قام راح للقاضى

جنازة وناس يبكوه

قابل فى السكة

وساب ماله لاخوه

رجع فى رايه

علشان نتمسك بيه

العالم فيه ايه

والعين ما تشبع به

الكل باطل باطل

نفسك طمعانة فيه

بتجرى ورا ايه

تبقى تدوم ليه

ان دامت لحد

لو ربحت العالم

انا ايه ينفعنى

واعيش فى خصام

واخسر نفسى

اغرت ابونا آدم

الحية زمان

ويعيش فى ظلام

وخليته يقع

واهدى سبيلى

يا رب ارشدنى

وياك يحلى لى

وشق لى طريقى

ويمينك تسندنى

تختاره ايدك

اسرع و اعنى

يارب اسمعنى

على اخوك الانسان

ابكى يا انسان

المهموم و الحيران

التايه و التعبان

محتاج للامان

المحتاج لحنان

لمكون الاكوان

قلبه بيصرخ

وحيله مع الابرار

شوف المكار

ويحولهم اشرار

يدخل فكر الناس

من شر الغدار

يارب نجينا

نكون ويا الابرار

واحفظنا فى اسمك

يبيت فيها و يرتاح

لمقبرة راح

ويطلب بالحاح

ثلاثة ايام يصلى

ويكتب له النجاح

الرب يتحنن

ويحتاج لايضاح

القرار مش سهل

قواه وعزاه

ظهر له ملاك

صحبه وصار معاه

ارشده للبرية

فرح لما راه

وصلوا المكان

واتغطى بسماه

هنا نام على الصخر

صوم وصلاة ودموع

بدا جهاده

وسجود بخشوع

نسك وطهارة

ويصبر على الجوع

يتحمل البرد

ويشرب من الينبوع

ياكل نصف خبزه

فمك ما داق طعام

مت عن العالم

ناظرةلله دايما

عينك ما شافت انسان

ما تركته لحظة

ورجليك الطاهرة

الارض نالت بركة

حتى اتقال عنها

على جسدك النحيف

لبست ثوب الليف

عشت غريب و ضيف

وصار حمله خفيف

بصلوات البار

ظلام المغارة انار

سيرة مملوءة فخار

تشهد لك بوقار

يا ايليا الجديد

يا موسي فى البرية

يا يوحنا العتيد

يا داود الوديع

نلت الكنز الاكيد

تركت غنى العالم

كل يوم يزيد

كان حبك لالهك

ملا الدنيا افراح

بخور طيبك قد فاح

ولا ندى يحيي الارواح

لولاك ما نزل مطر

يروى الارض براح

وفى اوانه النيل يفيض

يحتار فيه المداح

ومعاك الخير يجى

يظهر برك وضياك

لما اراد الله

انبا انطونيوس هناك

ارسل ملاكه للاب

بيطلب رؤياك

مشى لك يوم بتمام

وقفلت الباب وراك

فدخلت المغارة

وقال لك با مبارك

صرخ الانبا انطونيوس

اقبلنى لاتبارك

قبلت اليك وحوش

طلبت فوجدت

سالت فاعطيت

اؤمن انه يفتح لى

قرعت وانا

جه غراب و معاه

لما جاء الغروب

لضيف الانبا بولا

خبزايا كاملة

لك دالة عند الله

قال له انت حبيبه

يرسل نصف خبزة

لى سبعين سنة

ان الوقت قد حان

لما عرف القديس

والتعب والاحزان

يترك العالم

ارجع ديرك الان

قال للانبا انطونيوس

حامى الايمان

هات توب اثناسيوس

رجع بتوب اليقين

وهو فى الطريق

ولروح البار حاملين

شاف ملايكة تتهلل

بعد جهاد سنين

نال ابونا الاكليل

وبكى عليه بحنين

وكفنه بالثوب

اتبارك من جسده

لما اراد دفنه

سجدوا امام جسده

وحالا جاء نمرين

من بعد الاستئثان

وحفروا له مكان

وما احتجت لانسان

على ربك اتكلت

عشت على الارض ملاك

طوباك طوباك طوباك

لما شاف قلبك

الرب اختارك

وحوش تخضع بسرور

خدمتك الطيور

اسمك صار مزمور

دفنتك النمور

الرب بخرستوس

طوباك يا جبيب ايسوس

اللى اعطاك القدوس

ربحت العشر وزنات

كان عندك ايمان

الوزنة الاولى

يحفظ حياتك فى امان

ان الرب الديان

رجاء مع اطمئنان

الوزنة الثانية

زى ما عال شعبه زمان

الرب يعولك

كاملة و محتجبة

الثالثة محبة

كبرت حبة بحبة

كانت الاول حبة

عيشة بين الجبال

الرابعة احتمال

ده من المحال

مين بس اللى قال

حيوان ابدا و خشيت

الخامسة ما رايت

نلت نعمة و قويت

لكنك صليت

على الطبيعة كمان

السادسة سلطان

بحب الهك شبعان

يا بتول طاهر

وصايا الرب و خاضع

السابعة سامع

جسدك كله خاشع

وايدك رافع

وللذات سالب

الثامنة هارب

على كل التجارب

وبيسوع غالب

تركت لاجل الملك

التاسعة ممالك

وعلى قلبك مالك

فى طريقه سالك

ما حلاكى يا وحدة

الوزنة العاشرة

لذيذة وطعمة

والثمرة الحلوة

العيشة فى حاله

من ذاقك تحلى له

ويرتاح باله

تتحسن احواله

فى السما ويا الهك

طوباك طوباك طوباك

فرحان مع سيدك

نلت النعيم هناك

اولادك محتاجين

انبا بولا يا امين

تحرسهم كل حين

صلواتك عنهم

يا ساكن المغارة

طوباك با انبا بولا

بقيت لينا منارة

صارت سماء ثانية

سيرتك نقية

سيرتك شهية
