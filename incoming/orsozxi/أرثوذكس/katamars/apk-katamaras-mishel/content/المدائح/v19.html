الشهيدة يوستينا-هناك في أورشليم

حول العرش العظيم

هناك في أورشليم

حول يسوع فرحانين

ملايين القديسين

أطلبوها تكون معكم

أحدثكم عن واحدة منهم

تي مِنريت يوستينا

بصلواتها تشملكم

في الكنيسة مُستَديمة

عذراء شابة جميلة

تي مِنريت يوستينا

حياتها مستقيمة

ولم تَتَرجَّى سِواه

ذاقَت محبة الله

تي مِنريت يوستينا

قلبها دائم معاه

بِجمالِها الباهِر

رآها ابن الأكابِر

تي مِنريت يوستينا

وقلبها الطاهِر

وفي المال يغرِقها

طلب أن يتزوّجها

تي مِنريت يوستينا

والذهب يُلبِسها

وعِبادة أوثانه

رفضت أمواله

تي مِنريت يوستينا

فاشتعلت نيرانه

وبالتعذيب يتوعَّدها

فأخذ يهددها

تي مِنريت يوستينا

فرفعت ليسوع يديها

من المغرِب كان ماهِر

كبريانوس الساحِر

تي مِنريت يوستينا

جاء بسِحره يتفاخَر

عن الساحر الماهر

فسمع ابن الأكابر

تي مِنريت يوستينا

أنه في البلد حاضِر

وعن حاله اشتكى إليه

في الحال ذَهَبَ إليه

تي مِنريت يوستينا

ليميل يوستية إليه

أن يُبلِغه أمله

كبريانوس قد وعده

تي مِنريت يوستينا

ويوستينة تخضع له

فعادوا خائبين

أرسل لها قوة شياطين

تي مِنريت يوستينا

بالخِزي مُنهَزِمين

أمامِك مُنْكَسِرين

جنود الشياطين

تي مِنريت يوستينا

بصلاتِك كل حين

مع الشياطين الأشرار

كبريانوس فكَّر واحتار

تي مِنريت يوستينا

وأعطاهم إنذار

وأمامي تطرحوها

إن لم تهزِموها

تي مِنريت يوستينا

سأعبُد فاديها

أن يقترِب من المكان

لم يقدِر الشيطان

تي مِنريت يوستينا

بصلاتها كل آن

وعمل شكله زيها

لَبِسَ النَّجِس لبسها

تي مِنريت يوستينا

ففتح كِبريانوس لها

ونادىَ باسمها كمان

استقبلها وهو فرحان

تي مِنريت يوستينا

فصار الشيطان دخان

مجرَّد اسم يوستينا

فصرخ يا للحيرة

تي مِنريت يوستينا

يجعل إبليس بلا قيمة؟!

وندم على سِحره

فقام وأحرَق كتبه

تي مِنريت يوستينا

وعبد يسوع ربّه

ثوب الرَّهبنة ألبَسه

بابا أنطاكية عمَّده

تي مِنريت يوستينا

شمّاسًا فَقِسًّا رسمه

فاصطحَب يوستينا

ونال الأسقفية

تي مِنريت يوستينا

في قِرطاجنّة الإيبارشية

حوَّطتهم بالصلوات

صارَت أمًا للراهِبات

تي مِنريت يوستينا

فصار منهنَّ قدّيسا

ذاقّت معه الموت

في 21 توت

تي مِنريت يوستينا

ودخل كِلاهما الملكوت

لكن بحُبِّك لفادنا

لم تَعِظي يا يوستينة

تي مِنريت يوستينا

صِرتِ قدوة لينا

في صلواتِك اذكرينا

يوستينة إشفعي فينا

تي مِنريت يوستينا

الصلاة الدائِمة علينا
