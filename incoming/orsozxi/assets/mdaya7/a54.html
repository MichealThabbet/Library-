الشهيدة إيريني

المتعالي في سماه

+ لأمجد الإله

تي آجيا إيريني

وامدح عذراء فتاة

أبوها هو الوالي

+ أصلها عريق عالي

تي آجيا إيريني

وجمالها متلالي

لأبويها معزية

+ أميرة مقدونية

تي آجيا إيريني

وحيدة ونقية

فأسرعوا وبنوا لها

+ أحبها أبواها

تي آجيا إيريني

برج حصين يحميها

لك متع المسكونة

+ هنا عيشي مصونة

تي آجيا إيريني

وفتيات للمعونة

فخر لعلماء الجيل

+ ومعلم لك أصيل

تي آجيا إيريني

إكرميه واعطيه تبجيل

تهافتت عليها الرؤوس

+ كبرت تلك العروس

تي آجيا إيريني

قصدها أبوها ليكيتيوس

أولئك كبار بلدان

+ هؤلاء أمراء حسان

تي آجيا إيريني

فاختاري زين الشبان

فأعلم من أختار

+ يكفي أسبوع أتدبر

تي آجيا إيريني

لأعيش في سعادة أوفر

والعالم من أنشاه

+ قصري أبويا بناه

تي آجيا إيريني

دا عظيم هو الإله

إذا وقع ينكسر

+ لا يكون أبدًا حجر

تي آجيا إيريني

من يعبده يتعثر

برؤية تفسر بحساب

+ أجابها خالق الألباب

تي آجيا إيريني

حمامة ونسر وغراب

ستعرفي القدير

+ فسرها لها الخبير

تي آجيا إيريني

ورسوله لك ينير

وأعطاها مناها

+ ملاك الله عزاها

تي آجيا إيريني

رسول يسوع يلقاها

وقف أمامها يبوح

+ تيموثاوس حمله الروح

تي آجيا إيريني

بسر إلهنا المجروح

وأبيه والروح الهادي

+ أؤمن أنا بالفادي

تي آجيا إيريني

عمدني لجهادي

وثَبَتَتْ في المجروح

+ وًلِدَتْ تاني بالروح

تي آجيا إيريني

قَبِلَتْ صليب المذبوح

والحياة الأبدية

+ عَلّمَها المسيحية

تي آجيا إيريني

والدنيا سراب فانية

وتَحَصنَتْ بالإيمان

+ ملاها خوف الديان

تي آجيا إيريني

فحَطمَتْ كل الأوثان

والدها ليراها

+ بعد المهلة وافاها

تي آجيا إيريني

لتبوح بمشتهاها

ومباهجه لثواني

+ العالم كله فاني

تي آجيا إيريني

المسيحية هي إيماني

الساكن في أعلا سماه

+ عريسي هو الإله

تي آجيا إيريني

به أفرح وأتمناه

وكل ضعيف وعنيد

+ هذا دين العبيد

تي آجيا إيريني

للآلهة لي التمجيد

مًزَرْكَشَة بمهارة

+ آلهتَكْ حجارة

تي آجيا إيريني

تَابِعْها في خسارة

بين الخيل ضعوها

+ بغضب قال اربطوها

تي آجيا إيريني

لتموت فلا ترحموها

وأحيته بصلاتها

+ فَرَس صَرَعَ أباها

تي آجيا إيريني

فسجد لمن هداها

واستهان بجاه السلطان

+ ترك ظلام الأوثان

تي آجيا إيريني

وتبع نور الإيمان

متهللة بيسوع

+ مدة طويلة في الجوع

تي آجيا إيريني

طالبة إيمان الجموع

أو دفع مياه الأنهار

+ لم يرعبها المنشار

تي آجيا إيريني

بالإهانة صار لها وقار

سحبوها بقسوة لتصير

+ بقوة داست المسامير

تي آجيا إيريني

عبرة لتابعي القدير

فتمتعوا بالفردوس

+ كم أهدتْ لـه نفوس

تي آجيا إيريني

دي مضيئة كالشموس

ثَبَّت معها الجموع

+ تيموثاوس رسول يسوع

تي آجيا إيريني

للعالم لا رجوع

نالت الشهادة بيقين

+ في مسرى الحادي والعشرين

تي آجيا إيريني

بين خورس السمائيين

لآلاف من الشجعان

+ صرتِ أمًا في الإيمان

تي آجيا إيريني

تركوا دنس الأوثان

في الروحانية قيثارة

+ السلام لكِ يا مختارة

تي آجيا إيريني

تجذبيني كصنارة

بخشوع نصلي القداس

+ فلنمجد ماسياس

تي آجيا إيريني

ونقول لكِ أكسياس
