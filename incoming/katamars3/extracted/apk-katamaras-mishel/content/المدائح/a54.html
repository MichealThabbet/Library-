الشهيدة إيريني

المتعالى فى سماه

لأمجد الإله

تى أجيا أيرينى

وامدح عذراء فتاة

أبوها هو الوالى

أصلها عريق عالى

تى أجيا أيرينى

وجمالها متلالى

لأبويها معزية

أميرة مقدونية

تى أجيا أيرينى

وحيدة ونقية

فأسرعوا وبنوا لها

أحبها أبواها

تى أجيا أيرينى

برج حصين يحميها

لك متع المسكونة

هنا عيشى مصونة

تى أجيا أيرينى

وفتيات للمعونة

فخر لعلماء الجيل

ومعلم لك أصيل

تى أجيا أيرينى

إكرميه واعطيه تبجيل

تهافتت عليها الرؤوس

كبرت تلك العروس

تى أجيا أيرينى

قصدها أبوها ليكيتيوس

أولئك كبار بلدان

هؤلاء أمراء حسان

تى أجيا أيرينى

فاختارى زين الشبان

فأعلم من أختار

يكفى أسبوع أتدبر

تى أجيا أيرينى

لأعيش فى سعادة أوفر

والعالم من أنشاه

قصرى أبويا بناه

تى أجيا أيرينى

دا عظيم هو الإله

إذا وقع ينكسر

لا يكون أبدًا حجر

تى أجيا أيرينى

من يعبده يتعثر

برؤية تفسر بحساب

أجابها خالق الألباب

تى أجيا أيرينى

حمامة ونسر وغراب

ستعرفى القدير

فسرها لها الخبير

تى أجيا أيرينى

ورسوله لك ينير

وأعطاها مناها

ملاك الله عزاها

تى أجيا أيرينى

رسول يسوع يلقاها

وقف أمامها يبوح

تيموثاوس حمله الروح

تى أجيا أيرينى

بسر إلهنا المجروح

وأبيه والروح الهادى

أؤمن أنا بالفادى

تى أجيا أيرينى

عمدنى لجهادى

وثبت فى المجروح

ولدت تانى بالروح

تى أجيا أيرينى

قبلت صليب المذبوح

والحياة الأبدية

علمها المسيحية

تى أجيا أيرينى

والدنيا سراب فانية

وتحصنت بالإيمان

ملاها خوف الديان

تى أجيا أيرينى

فحطمت كل الأوثان

والدها ليراها

بعد المهلة وافاها

تى أجيا أيرينى

لتبوح بمشتاهها

ومباهجه اثوانى

العالم كله فانى

تى أجيا أيرينى

المسيحية هى إيمانى

الساكن فى أعلا سماه

عريسى هو الإله

تى أجيا أيرينى

به أفرح وأتمناه

وكل ضعيف وعنيد

هذا دين العبيد

تى أجيا أيرينى

للألهة لى التمجيد

مزركشة بمهارة

آلهتك حجارة

تى أجيا أيرينى

تابعها فى خسارة

بين الخيل ضعوها

بغضب قال اربطوها

تى أجيا أيرينى

لتموت فلا ترحموها

وأحيته بصلاتها

فرس صرع أباها

تى أجيا أيرينى

فسجد لمن هداها

واستهان بجاه السلطان

ترك ظلام الأوثان

تى أجيا أيرينى

وتبع نور الإيمان

متهللة بيسوع

مدة طويلة فى الجوع

تى أجيا أيرينى

طالبة إيمان الجموع

أو دفع مياه الأنهار

لم يرعبها المنشار

تى أجيا أيرينى

بالإهانة صار لها وقار

سحبوها بقسوة لتصير

بقوة داست المسامير

تى أجيا أيرينى

عبرة لتباعى القدير

فتمتعوا بالفردوس

كم أهدت لـه نفوس

تى أجيا أيرينى

دى مضيئة كالشموس

ثبت معها الجموع

تيموثاوس رسول يسوع

تى أجيا أيرينى

للعالم لا رجوع

نالت الشهادة بيقين

فى مسرى الحادى والعشرين

تى أجيا أيرينى

بين خورس السمائيين

لآلاف من الشجعان

صرتِ أمًا فى الإيمان

تى أجيا أيرينى

تركوا دنس الأوثان

فى الروحانية قيثارة

السلام لكِ يا مختارة

تى أجيا أيرينى

تجذبينى كصنارة

بخشوع نصلى القداس

فلنمجد ماسياس

تى أجيا أيرينى

ونقول لك أكسياس
