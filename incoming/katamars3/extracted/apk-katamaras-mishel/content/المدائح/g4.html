القديس الانبا رويس-أفتح فاي بإسم الله

واحد لا رب سواه

افتح فاي باسم الله

القديس انبا رويس

وامدح قديس اصطفاه

امدح هذا الانسان

بدء كلامي يا اخوان

القديس انبا رويس

ذكره علي كل لسان

بالغربية بيقين

مولود في منية يمين

القديس انبا رويس

هذا الرجل الامين

وامه سارة يا رفاق

ابوه هو اسحق

القديس انبا رويس

مؤمنين بالرب الخلاق

فريج ودعياه

لما ولد اسمياه

القديس انبا رويس

وهو ناظر الاله

في الفلاحة مع والده

وكان يعمل من صغره

القديس انبا رويس

ويبيع الملح علي جمله

يحرك الرأس كثير

جمله قاعود صغير

القديس انبا رويس

فدعي رويس بتدبير

بضيقة للمسيحيين

اتي الشيطان اللعين

القديس انبا رويس

فجاء الي مصر الامين

في مصر يا اخوة

ولم يكن له مأوي

القديس انبا رويس

ويبيع الملح بالبركة

في صلاته لربه

قضي اغلب وقته

القديس انبا رويس

ساهر طول ليله

او ثوبا يا احباء

لم يقتني رداء

القديس انبا رويس

عاش فقر اله السماء

ومن الروح ملآنا

وكان عريانا

القديس انبا رويس

تقوي وايمانا

ومنظره يا مختاري

وكان رأسه عاري

القديس انبا رويس

كسواح البراري

من كثرة البكاء

وعيناه حمراء

القديس انبا رويس

وهو من الانقياء

كنذير لربه

ولم يحلق شعره

القديس انبا رويس

الهه من احبه

هذا الرجل الهمام

وكان قليل الكلام

القديس انبا رويس

مملوء من السلام

الي السماء رفعه

يسوع قد اخذه

القديس انبا رويس

وناوله بيده

ضربه بقسوة يا حضار

واحد من الاشرار

القديس انبا رويس

لم يفتح فمه البار

الرجل الطوباوي

كان حاضر يا اخوتي

القديس انبا رويس

انبا مرقص الانطوني

كان ينادي ام الهه

في آخر حياته

لقديس انبا رويس

ويقول في صلواته

من همي خلصيني

يا عذراء خذيني

القديس انبا رويس

من العالم نجيني

وانا قلبي ذليل

الحمل علي ثقيل

القديس انبا رويس

اعطيني فرح وتهليل

قد رفضوا باصرار

لان الناس الاشرار

القديس انبا رويس

سماع كلام البار

البابا متاؤوس

قد عاصر المأنوس

القديس انبا رويس

خليفة مارمرقس

انه حبس نفسه

ومن غريب امره

القديس انبا رويس

داخل مخزن وحده

فاتاه عمانوئيل

عند تلميذه ميخائيل

القديس انبا رويس

اعطاه السلام والتهليل

في ذلك المكان الامين

واقام تسع سنين

القديس انبا رويس

حتي تنيح بيقين

صعدت روحه بمهابة

في واحد وعشرين بابه

القديس انبا رويس

ونال السعادة

اشفية وآيات

كم صنع من معجزات

القديس انبا رويس

لكل غاد وآت

شفاعته ترفعنا

صلواته تكون معنا

القديس انبا رويس

للرب فيقبلنا

من كل الدسائس

وتكون لنا حارس

القديس انبا رويس

وللملكوت نرث

كل المؤمنين

تفسير اسمك في افواه

انبا رويس اعنا اجمعين

الكل يقولون يا اله
