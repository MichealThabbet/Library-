الشهيدة فبرونيا الراهبة

ونُسبحه فى أعلى سماه

نُمجد الإله

تى آجيا فبرونيا

ونَمدح عذراء فتاه

صغيرة فى بهاء

فبرونيا كانت عذراء

تى آجيا فبرونيا

أحبت ملك السماء

فى صلاة كل الأوقات

عاشت مع الراهبات

تى آجيا فبرونيا

كالعذارى الحكيمات

فى العالم ضيق ودموع

قال الرب يسوع

تى آجيا فبرونيا

قلبها للسماء مرفوع

الجند والسلطان

أهاج الشيطان

تى آجيا فبرونيا

على أولاد الإيمان

والرب إله القوات

قتل وسجن الراهبات

تى آجيا فبرونيا

ينظر من السموات

على دير راهبات أبرار

هجم الجند الأشرار

تى آجيا فبرونيا

قتلوا ونهبوا الديار

قالوا دى هدية ثمينة

رأوا راهبة جميلة

تى آجيا فبرونيا

للخليفة ذو القيمة

انجدنى يا ربى

فبرونيا تُصلى

تى آجيا فبرونيا

وأحفظ طهارتى

واعمل معى جميل

قالت اتركنى يا أمير

تى آجيا فبرونيا

أعطيك سلاح جليل

لا تؤثر فيه السيوف

زيت ينجى النفوس

تى آجيا فبرونيا

ينفع فى الحروب

وبالزيت ادهنى

قال لها جربى

تى آجيا فبرونيا

رقبتكِ وأرينى

البتول بحكمتها

دهنت رقبتها

تى آجيا فبرونيا

والجنود تنظرها

ولم يَعلم بالحيلة

ضرب رأس الصغيرة

تى آجيا فبرونيا

وفازت الأمينة

وحفظت بتوليتها

انفصلت رأسها

تى آجيا فبرونيا

ونالت أكاليلها

إيمانك كالجبال

طوباك يا مثال

تى آجيا فبرونيا

قدوة للأجيال

يا طاهرة يا عفيفة

أذكرينا يا قديسة

تى آجيا فبرونيا

يا منارة فى الكنيسة

كل المؤمنين

تسبيح أسمَك فى افواه

القديسة فبرونيا أعنَّا أجمعين

الكل يقولون يا إله
