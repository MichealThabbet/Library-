الشهيدة إفرونيا العفيفة

النقية المختارة

السلام لفخر العذارى

تى آجيا افرونيا

افرونيا كنز الطهارة

فى حياتها مستقيمة

عذراء شابة جميلة

تى آجيا افرونيا

عاشت يتيمة

أوريانة خالتها

فقدت والديها

تى آجيا افرونيا

بالحنان تراعيها

فى دير القديسين

هناك بين النهرين

تى آجيا افرونيا

بخوف الله عايشين

تحسن كل تدبير

أوريانه رئيسة الدير

تى آجيا افرونيا

عاشت فى نقاء وخير

بخشوع فى المطانيات

بدموع فى الصلوات

تى آجيا افرونيا

بجهاد ومشقات

والأعداء والتهديد

هاج عليهن العنيد

تى آجيا افرونيا

معهُنَّ الرب أكيد

بالظلم عذبوها

أوريانة أهانوها

تى آجيا افرونيا

عجوز لم يرحموها

اتركوها ولا تتركونى

افرونيا قالت خذونى

تى آجيا افرونيا

بقوتكم لم تخيفونى

وبقسوة لا تُزال

قيدوها بالحبال

تى آجيا افرونيا

وعدوها بالآمال

السلطان والوجود

رفضت كل الوعود

تى آجيا افرونيا

مهما كانت القيود

عذَّبوها بلا شفقة

عصروها بلا رحمة

تى آجيا افرونيا

أنا ليسوع عبدة

وتمزيق ملابسها

أرادوا كشف جسدها

تى آجيا افرونيا

وتشويه جمالها

وكسروا لها الأسنان

قطعوا لها اللسان

تى آجيا افرونيا

حرقوها بالنيران

ومسح لها الدموع

أتى إليها يسوع

تى آجيا افرونيا

صار الحب ينبوع

والملائكة أخذوها

أخيرًا ذبحوها

تى آجيا افرونيا

والإكليل ألبسوها

على صغر سنواتك

عظيم هو إيمانك

تى آجيا افرونيا

حققتى آمالك

ذهبتِ للحبيب

فى واحد من أبيب

تى آجيا افرونيا

يفوح منكِ الطيب

فى صلاتك اُذكرينا

افرونيا اشفعى فينا

تى آجيا افرونيا

علمينا حُب فادينا

كل المؤمنين

تسبيح أسمَك فى أفواه

الشهيدة افرونيا أعنا أجمعين

الكل يقولون يا إله
