السـلام لقديس عــظـيـــم
كــامل طـــاهر حـــكيـــم
يليـــق لــــه الـتـكـريـــم
السلام للشـهيد أبــــانــــوب
ولــد بنهيســة
مــن مريــــم الـقديســـة
وأبـــوه مقـاره
السلام للشهيد أبــــانــــوب
بعـــد 12 سنة
إنتقــلا للســما
وتركـــوه فــي صبــــــاه
السلام للشهيد أبــــانــــوب
ذهــب لكنيسـة الابــــــــرار
ليتنــاول مــن الاســـــــرار
فــي وداعــة ووقـــــــار
السلام للشهيد أبــــانــــوب
فــي عـظـــة القـــــداس
سمــع عـــن دقلديــانوس
وإضطـهادة لشـعب إيسوس
السـلام للشهيد أبــــانـــوب
ذهب القديــس وبــــــــاع
الذهب والمتاع
وأعطاه للجياع
السلام للشهيد أبــــانــــوب
وصلي الي الله
يطلــب مكــان الــــــولاة
لينــال أكليــل الشــــهادة
السلام للشهيد أبـــانـــــوب
ظــهــر لــــه ميخــــــائيل
مــــن قبـــــل عمانـــــوئيل
ووعـــــده بالأكـــاليل
السلام للشهيد أبـــانــــوب
أمـــر الـــوالي بجلـــــــده
أربــع مـــرات عـلي بطنـــه
فخرجــــت أحشــــاؤه
السلام للشهيد أبــــانــوب
أتــي المــلاك وشفــــــاه
ومــن الــوالـي نجــــــــاه
وقـام بكل قواه
السلام للشهيد أبـــانــــوب
صــار الـــوالي كـــالمجنون
وقال القوه في الســـجـون
مع إخوانــه المســيحيون
السلام للشهيد أبـــانـــــوب
شـجع كــل المسـجونين
فتمســـكوا بالــــدين
وقــالوا نحــن مســـــيحيين
السلام للشهيد أبـــانـــــوب
أمــر الـــوالي بتجميعــــهم
وقتلهم باجمعهم
أكــثر من ألف عـــددهــــم
السلام للشهيد أبـــانـــوب
فعلـــق أبانوب
علـي صــاري الــمركـــــب
وإتجـــه إلـــي أتـريـــــب
السلام للشهيد أبـــانــــوب
نزف أنفه وفاه
فــأتى المــلاك وشـــــــفاه
وفــك رباطـاته
السلام للشهيد أبـــانـــــوب
شُـــلَ والــى سمنــــــــود
وعُمــىَ الجنود
فــآمنوا بالحي الموجـــــود
السلام للشهيد أبـــانــــــوب
أعلنوا إيمانهم
فعفــي الـــرب عـنــــــــهم
بصلاة قديسهم
السلام للشهيد أبــانــــــوب
واعـــــــترفوا بالمســــــيح
أنـــه الإلــــه الصحيــــــح
الــــذي فـيــه نســـــــتريح
السلام للشهيد أبـــانـــــــوب
أغتــاظ والي أتريـــــــب
من إيمانـهم العجيـــــب
فقطعـــهم إرب إرب
السلام للشهيد أبــانــــــوب
وأتـي بـالقديس ودعــــــــــاه
أن يقـف وســط الــــــــــولاة
فلـــم يخشــي أمــــــــــــره
السلام للشهيد أبـــانــــوب
فأمروا بتعذيبه
بكـــيَ جسده
وحَـــــــرق أعضــــاؤه
السلام للشهيد أبـــانــــوب
فـــأتي يســوع وشـــــــــفاه
ومــن الحــرق نجـــــــــــاه
وبالملائكــــة قــــــــــواه
السلام للشهيد أبــانــــــوب
وقطعـوا رجليه
وأيضاَ يديـــه
ليقضـــوا عليه
السلام للشهيد أبــانـــــــوب
أتـــي المـــلاك وشــــــــــفاه
ومــن الــوالي نجـــــــــــــاه
وقام بكل قواه
السلام للشهيد أبــانـــــــوب
وبعــد عذابـات شــــديــدة
وقسـوة مريرة
نـــال إكليـــل الشــــــهادة
السلام للشهيد أبـــانـــــوب
تفسير إسمك فـي أفــــواه
كل المؤمنين
الكل يقولون يا إلـــــــه
